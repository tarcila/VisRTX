cmake_minimum_required(VERSION 3.17)
project(principled_volume_data LANGUAGES CXX)
add_executable(write_nvdb write_nvdb.cpp)
target_compile_features(write_nvdb PRIVATE cxx_std_17)
target_include_directories(write_nvdb PRIVATE ../../devices/rtx/external/nanovdb/include)
option(BUILD_QUALITY_BASELINE "Build the public-ANARI Quality measurement driver" OFF)
if(BUILD_QUALITY_BASELINE)
  find_package(anari REQUIRED)
  find_package(CUDAToolkit REQUIRED)
  add_executable(quality_baseline quality_baseline.cpp)
  target_compile_features(quality_baseline PRIVATE cxx_std_17)
  target_include_directories(quality_baseline PRIVATE ../../devices/rtx/external/nanovdb/include)
  target_link_libraries(quality_baseline PRIVATE anari::anari CUDA::cudart)
endif()
option(EXPORT_OPENVDB "Also write Blender-compatible OpenVDB files" OFF)
if(EXPORT_OPENVDB)
  find_package(OpenVDB REQUIRED)
  target_link_libraries(write_nvdb PRIVATE OpenVDB::openvdb)
  target_compile_definitions(write_nvdb PRIVATE EXPORT_OPENVDB)
endif()
